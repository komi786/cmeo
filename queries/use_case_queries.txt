Competency question queries for the CMEO knowledge graph
Updated 2026-10-14 for CMEO release 2026-10-14.

All ontology terms are now real OBO / SIO IRIs (obo:XXX_0000000). Earlier versions used
made-up IRIs such as obi.owl/protocol or cmeo:has_value, which no longer match the graph.
The prefix cmeo: is only used for instance data and KG-local terms that have no ontology class.

Common prefixes (repeated in every query so each one runs on its own):

CQ1: What are the protocol specifications of the randomized controlled trials?

PREFIX obo:  <http://purl.obolibrary.org/obo/>
PREFIX dc:   <http://purl.org/dc/elements/1.1/>

SELECT ?study_name
  (GROUP_CONCAT(DISTINCT STR(?part_uri); separator=";") AS ?all_parts_uri)
  (GROUP_CONCAT(DISTINCT STR(?part_value); separator=";") AS ?all_parts_value)
WHERE {
  GRAPH <https://w3id.org/CMEO/graph/studies_metadata> {
    ?study a obo:OBCS_0000107 ;                 # randomized controlled trial design
           obo:BFO_0000051 ?protocol ;          # has part
           dc:identifier ?study_name .
    ?protocol a obo:OBI_0000272 ;               # protocol
              obo:BFO_0000051+ ?part_uri .
    ?part_uri obo:STATO_0000129 ?part_value .   # has value
  }
}
GROUP BY ?study_name


CQ2: Which studies investigated heart failure condition?

PREFIX obo:  <http://purl.obolibrary.org/obo/>
PREFIX dc:   <http://purl.org/dc/elements/1.1/>
PREFIX rdfs: <http://www.w3.org/2000/01/rdf-schema#>

SELECT DISTINCT ?study_name ?description
WHERE {
  GRAPH <https://w3id.org/CMEO/graph/studies_metadata> {
    ?study_design dc:identifier ?study_name ;
                  obo:IAO_0000136 ?descriptor ;   # is about
                  obo:BFO_0000051 ?prot .
    ?descriptor rdfs:label ?description .
    ?prot a obo:OBI_0000272 ;
          obo:BFO_0000051 ?obj_spec .
    ?obj_spec a obo:IAO_0000005 ;                 # objective specification
              obo:STATO_0000129 ?obj_val .
    ?prot obo:BFO_0000051 ?ec .
    ?ec a obo:OBI_0500026 ;                       # eligibility criterion
        obo:BFO_0000051 ?inc .
    ?inc a obo:OBI_0500027 ;                      # inclusion criterion
         obo:BFO_0000051 ?spec_inc .
    ?spec_inc obo:STATO_0000129 ?inc_value .
    FILTER(CONTAINS(?obj_val, "heart failure") || CONTAINS(?inc_value, "heart failure"))
  }
}


CQ3: What are the population characteristics in different studies?

PREFIX obo:  <http://purl.obolibrary.org/obo/>
PREFIX dc:   <http://purl.org/dc/elements/1.1/>
PREFIX rdfs: <http://www.w3.org/2000/01/rdf-schema#>

SELECT ?study_name
  (GROUP_CONCAT(DISTINCT LCASE(STR(?characteristic_value)); separator="; ") AS ?qualities)
  (GROUP_CONCAT(DISTINCT STR(?characteristic_type); separator="; ") AS ?quality_types)
WHERE {
  GRAPH <https://w3id.org/CMEO/graph/studies_metadata> {
    ?study_design dc:identifier ?study_name ;
                  obo:BFO_0000051 ?prot .
    ?prot a obo:OBI_0000272 ;
          obo:BFO_0000051 ?ec .
    ?ec a obo:OBI_0500026 ;
        obo:RO_0000058 ?hse_uri .                 # is concretized as (human subject enrollment)
    ?hse_uri obo:RO_0002234 ?population .         # has output
    ?population a obo:OBI_0000181 ;               # population
                obo:RO_0000053 ?quality .         # has characteristic
    OPTIONAL { ?quality rdfs:label ?characteristic_value }
    OPTIONAL { ?quality a ?characteristic_type }
  }
}
GROUP BY ?study_name
LIMIT 3


CQ4: Which studies match the PICO criteria?

PREFIX obo:  <http://purl.obolibrary.org/obo/>
PREFIX dc:   <http://purl.org/dc/elements/1.1/>
PREFIX xsd:  <http://www.w3.org/2001/XMLSchema#>

SELECT DISTINCT ?study_name
  (GROUP_CONCAT(DISTINCT CONCAT(LCASE(COALESCE(STR(?diagnosis_value), "")), " [", COALESCE(STR(?min_val), "?"), "-", COALESCE(STR(?max_value), "?"), "]"); separator="; ") AS ?population)
  (GROUP_CONCAT(DISTINCT LCASE(STR(?intervention_value)); separator="; ") AS ?interventions)
  (GROUP_CONCAT(DISTINCT LCASE(STR(?comparator_value)); separator="; ") AS ?comparators)
  (GROUP_CONCAT(DISTINCT LCASE(STR(?outcome_value)); separator="; ") AS ?outcomes)
WHERE {
  GRAPH <https://w3id.org/CMEO/graph/studies_metadata> {
    ?study_design dc:identifier ?study_name ;
                  obo:BFO_0000051 ?prot .
    ?prot a obo:OBI_0000272 ;
          obo:BFO_0000051 ?eligibility_criterion .
    ?eligibility_criterion a obo:OBI_0500026 ;
                           obo:BFO_0000051 ?inclusion_criteria .
    ?inclusion_criteria a obo:OBI_0500027 .

    # Diagnostic inclusion
    OPTIONAL {
      ?inclusion_criteria obo:BFO_0000051 ?diag_part .
      ?diag_part a obo:OBI_0002453 ;               # health status inclusion criterion
                obo:STATO_0000129 ?diagnosis_value .
      FILTER(CONTAINS(?diagnosis_value, "nyha class ii"))
    }
    # Age inclusion
    OPTIONAL {
      ?inclusion_criteria obo:BFO_0000051 ?age_part .
      ?age_part a obo:OBI_0002555 ;               # age group inclusion criterion
                obo:BFO_0000051 ?min_val_specification, ?max_value_specification .
      ?min_val_specification a obo:OBI_0002556 ;  # minimum age value specification
                             obo:STATO_0000129 ?min_val .
      ?max_value_specification a obo:OBI_0002557 ;  # maximum age value specification
                               obo:STATO_0000129 ?max_value .
      FILTER(xsd:decimal(?min_val) >= 60 && xsd:decimal(?max_value) <= 85)
    }
    # Intervention
    OPTIONAL {
      ?prot obo:BFO_0000051 ?intervention .
      ?intervention a obo:CMEO_0000072 ;          # intervention specification
                    obo:STATO_0000129 ?intervention_value .
      FILTER(CONTAINS(?intervention_value, "intensified"))
    }
    # Comparator
    ?prot obo:BFO_0000051 ?comparator .
    ?comparator a obo:CMEO_0000071 ;              # comparator specification
                obo:STATO_0000129 ?comparator_value .
    FILTER(CONTAINS(?comparator_value, "standard guided"))
    # Outcomes
    OPTIONAL {
      ?prot obo:BFO_0000051 ?outcome .
      ?outcome a obo:CMEO_0000063 ;               # study outcome specification
               obo:BFO_0000051 ?suboutcome_uri .
      ?suboutcome_uri obo:STATO_0000129 ?outcome_value .
      FILTER(CONTAINS(?outcome_value, "survival"))
    }
  }
}
GROUP BY ?study_name


CQ5: How are missing values encoded in study A?

PREFIX obo:  <http://purl.obolibrary.org/obo/>
PREFIX dc:   <http://purl.org/dc/elements/1.1/>

SELECT DISTINCT ?variable ?missing_value
WHERE {
  GRAPH <https://w3id.org/CMEO/graph/time-chf> {
    ?dataElement a obo:CMEO_0000009 ;             # data element
                 dc:identifier ?variable ;
                 obo:OBI_0001938 ?missingSpec .   # has value specification
    ?missingSpec a obo:CMEO_0000028 ;             # missing value specification
                 obo:STATO_0000129 ?missing_value .
  }
}
LIMIT 3


CQ6: Which categorical variables does time-chf include, and what are their distributions?

PREFIX obo:  <http://purl.obolibrary.org/obo/>
PREFIX dc:   <http://purl.org/dc/elements/1.1/>

SELECT ?variable ?fq_value
WHERE {
  GRAPH <https://w3id.org/CMEO/graph/time-chf> {
    ?dataElement a obo:CMEO_0000009 ;
                 dc:identifier ?variable .
    ?statistical_variable a ?type ;
                          obo:IAO_0000219 ?dataElement .   # denotes
    VALUES ?type { obo:STATO_0000090 obo:STATO_0000087 }   # dichotomous / polychotomous variable
    ?ds a obo:IAO_0000100 ;                                 # homogenous data collection
        obo:IAO_0000136 ?statistical_variable ;
        obo:OBI_0000295 ?eda_proc .                         # is specified input of
    ?eda_proc a obo:CMEO_0000073 ;                          # exploratory data analysis
              obo:OBI_0000299 ?stats .                      # has specified output
    ?stats a obo:STATO_0000039 ;                            # statistic
           obo:BFO_0000051 ?part_uri .
    ?part_uri a obo:OBCS_0000068 ;                          # frequency distribution
              obo:STATO_0000129 ?fq_value .
  }
}
LIMIT 3


CQ7: Which variables are similar between selected studies (time-chf and gissi-hf)?
(template: {graph_repo}, {source}, {target} are filled in by the code)
Each side is aggregated on its own and then joined on the OMOP id. The old version put both
sides in one GROUP_CONCAT; Oxigraph returns nothing for a GROUP_CONCAT that meets an unbound
value, so source/target columns came back empty.

PREFIX obo:  <http://purl.obolibrary.org/obo/>
PREFIX sio:  <http://semanticscience.org/resource/>
PREFIX dc:   <http://purl.org/dc/elements/1.1/>
PREFIX rdf:  <http://www.w3.org/1999/02/22-rdf-syntax-ns#>
PREFIX rdfs: <http://www.w3.org/2000/01/rdf-schema#>
PREFIX skos: <http://www.w3.org/2004/02/skos/core#>

SELECT ?omop_id
  (COALESCE(?src_lbl, ?tgt_lbl) AS ?code_label)
  (COALESCE(?src_val, ?tgt_val) AS ?code_value)
  (?src_de_uris AS ?source_de_uris)
  (?tgt_de_uris AS ?target_de_uris)
  (?src_definition AS ?source_definition)
  (?tgt_definition AS ?target_definition)
  (?src_domain AS ?source_domain)
  (?tgt_domain AS ?target_domain)
WHERE {
  {
    SELECT DISTINCT ?omop_id WHERE {
      { GRAPH <{graph_repo}/{source}> { ?de a obo:CMEO_0000009 ; skos:closeMatch/rdf:_1/obo:IAO_0000219/obo:STATO_0000129 ?omop_id . } }
      UNION
      { GRAPH <{graph_repo}/{target}> { ?de a obo:CMEO_0000009 ; skos:closeMatch/rdf:_1/obo:IAO_0000219/obo:STATO_0000129 ?omop_id . } }
    }
  }
  OPTIONAL {
      SELECT ?omop_id (SAMPLE(?lbl) AS ?src_lbl) (SAMPLE(?val) AS ?src_val)
        (GROUP_CONCAT(DISTINCT ?de_str; SEPARATOR="||") AS ?src_de_uris)
        (GROUP_CONCAT(DISTINCT ?combined; SEPARATOR="||") AS ?src_definition)
        (GROUP_CONCAT(DISTINCT ?dom; SEPARATOR="||") AS ?src_domain)
      WHERE {
        GRAPH <{graph_repo}/{source}> {
          ?de a obo:CMEO_0000009 ;                       # data element
              rdfs:label ?var_label ;
              dc:identifier ?var ;
              skos:closeMatch ?codeSet .
          ?codeSet rdf:_1 ?codeNode .
          ?codeNode rdfs:label ?lbl ;
                    obo:STATO_0000129 ?val ;             # has value
                    obo:IAO_0000219/obo:STATO_0000129 ?omop_id .
          OPTIONAL { ?de sio:SIO_000008/skos:closeMatch/rdfs:label ?vis_label . }   # has attribute (visit)
          OPTIONAL { ?de sio:SIO_000255/obo:STATO_0000129 ?dom_raw . }               # has annotation (domain)
          BIND(COALESCE(?dom_raw, "") AS ?dom)
          BIND(CONCAT(?var, ";;", ?var_label, ";;", COALESCE(?vis_label, "N/A")) AS ?combined)
          BIND(STR(?de) AS ?de_str)
        }
      }
      GROUP BY ?omop_id
  }
  OPTIONAL {
      SELECT ?omop_id (SAMPLE(?lbl) AS ?tgt_lbl) (SAMPLE(?val) AS ?tgt_val)
        (GROUP_CONCAT(DISTINCT ?de_str; SEPARATOR="||") AS ?tgt_de_uris)
        (GROUP_CONCAT(DISTINCT ?combined; SEPARATOR="||") AS ?tgt_definition)
        (GROUP_CONCAT(DISTINCT ?dom; SEPARATOR="||") AS ?tgt_domain)
      WHERE {
        GRAPH <{graph_repo}/{target}> {
          ?de a obo:CMEO_0000009 ;                       # data element
              rdfs:label ?var_label ;
              dc:identifier ?var ;
              skos:closeMatch ?codeSet .
          ?codeSet rdf:_1 ?codeNode .
          ?codeNode rdfs:label ?lbl ;
                    obo:STATO_0000129 ?val ;             # has value
                    obo:IAO_0000219/obo:STATO_0000129 ?omop_id .
          OPTIONAL { ?de sio:SIO_000008/skos:closeMatch/rdfs:label ?vis_label . }   # has attribute (visit)
          OPTIONAL { ?de sio:SIO_000255/obo:STATO_0000129 ?dom_raw . }               # has annotation (domain)
          BIND(COALESCE(?dom_raw, "") AS ?dom)
          BIND(CONCAT(?var, ";;", ?var_label, ";;", COALESCE(?vis_label, "N/A")) AS ?combined)
          BIND(STR(?de) AS ?de_str)
        }
      }
      GROUP BY ?omop_id
  }
}
ORDER BY ?omop_id

CQ9: What studies permit data access for research on congestive heart failure, subject to ethics approval?

PREFIX obo:  <http://purl.obolibrary.org/obo/>
PREFIX cmeo: <https://w3id.org/CMEO/>
PREFIX dc:   <http://purl.org/dc/elements/1.1/>
PREFIX rdfs: <http://www.w3.org/2000/01/rdf-schema#>

SELECT DISTINCT ?study_name
WHERE {
  GRAPH <https://w3id.org/CMEO/graph/studies_metadata> {
    ?assignment a cmeo:data_use_permission_assignment ;   # KG-local, no ontology class
                obo:IAO_0000136 ?x_study ;
                obo:OBI_0000299 ?data_policy , ?data_modifier .
    ?x_study a obo:OBI_0000471 ;                           # study design execution
             dc:identifier ?study_name .
    ?data_policy a obo:DUO_0000007 ;                       # disease specific research
                 obo:IAO_0000136 ?disease_code .
    ?data_modifier a obo:DUO_0000021 .                     # ethics approval required

CQ10: Retrieve variable profile of a group of variables
(template: {graph_repo}, {source}, {values_str} are filled in by the code)

PREFIX obo:  <http://purl.obolibrary.org/obo/>
PREFIX sio:  <http://semanticscience.org/resource/>
PREFIX dc:   <http://purl.org/dc/elements/1.1/>
PREFIX rdf:  <http://www.w3.org/1999/02/22-rdf-syntax-ns#>
PREFIX rdfs: <http://www.w3.org/2000/01/rdf-schema#>
PREFIX skos: <http://www.w3.org/2004/02/skos/core#>
PREFIX xsd:  <http://www.w3.org/2001/XMLSchema#>

SELECT ?identifier
  (SAMPLE(?_stat_label)    AS ?stat_label)
  (SAMPLE(?_unit_label)    AS ?unit_label)
  (SAMPLE(?_data_type_val) AS ?data_type_val)
  (MAX(?_min_v)            AS ?min_val)
  (MAX(?_max_v)            AS ?max_val)
  (GROUP_CONCAT(DISTINCT STR(?cat_omop_id); SEPARATOR="||") AS ?cat_omop_ids)
  (GROUP_CONCAT(DISTINCT ?_catL; SEPARATOR="||") AS ?all_cat_labels)
  (GROUP_CONCAT(DISTINCT ?_catV; SEPARATOR="||") AS ?all_original_cat_values)
  (SAMPLE(?_ordered_code_labels) AS ?code_label)
  (SAMPLE(?_ordered_code_values) AS ?code_value)
  (SAMPLE(?_ordered_omop_ids) AS ?omop_ids)
WHERE {
  GRAPH <{graph_repo}/{source}> {
    VALUES ?identifier { {values_str} }
    ?dataElement a obo:CMEO_0000009 ;
                 dc:identifier ?identifier .

    OPTIONAL { ?dataElement obo:IAO_0000235/obo:STATO_0000129 ?_stat_label . }   # denoted by

    OPTIONAL {
      ?dataElement sio:SIO_000008 ?dt .
      ?dt a obo:CMEO_0000059 ;                     # data type
          obo:STATO_0000129 ?_data_type_val .
    }

    BIND(URI(CONCAT(STR(?dataElement), "/statistic")) AS ?statUri)

    OPTIONAL {
      ?statUri obo:BFO_0000051 ?minPart .
      ?minPart a obo:STATO_0000150 ;               # minimum observed value
               obo:STATO_0000129 ?_min_v .
    }
    OPTIONAL {
      ?statUri obo:BFO_0000051 ?maxPart .
      ?maxPart a obo:STATO_0000151 ;               # maximum observed value
               obo:STATO_0000129 ?_max_v .
    }

    OPTIONAL { ?dataElement obo:IAO_0000039/skos:closeMatch/obo:STATO_0000129 ?_unit_label . }

    # Categories
    OPTIONAL {
      ?cat_val a obo:OBI_0001930 ;                 # categorical value specification
               obo:OBI_0001927 ?dataElement ;      # specifies value of
               obo:STATO_0000129 ?_catV .
      OPTIONAL {
        ?cat_val skos:closeMatch ?cat_code .
        ?cat_code a skos:Concept ;
                  rdfs:label ?_catL .
        OPTIONAL { ?cat_code obo:IAO_0000219/obo:STATO_0000129 ?cat_omop_id . }
      }
    }

    # Codes in sequence order
    OPTIONAL {
      SELECT ?dataElement
        (GROUP_CONCAT(?cL; SEPARATOR="||") AS ?_ordered_code_labels)
        (GROUP_CONCAT(?cV; SEPARATOR="||") AS ?_ordered_code_values)
        (GROUP_CONCAT(STR(?omop_id); SEPARATOR="||") AS ?_ordered_omop_ids)
      WHERE {
        GRAPH <{graph_repo}/{source}> {
          SELECT ?dataElement ?cL ?cV ?omop_id ?seqNum WHERE {
            ?dataElement skos:closeMatch ?codeSet .
            ?codeSet ?seqPred ?codeNode .
            ?codeNode a skos:Concept ;
                      rdfs:label ?cL ;
                      obo:STATO_0000129 ?cV ;
                      obo:IAO_0000219/obo:STATO_0000129 ?omop_id .
            FILTER(STRSTARTS(STR(?seqPred), CONCAT(STR(rdf:), "_")))
            BIND(xsd:integer(STRAFTER(STR(?seqPred), CONCAT(STR(rdf:), "_"))) AS ?seqNum)
          }
          ORDER BY ?dataElement ?seqNum
        }
      }
      GROUP BY ?dataElement
    }
  }
}
GROUP BY ?identifier
ORDER BY ?identifier
